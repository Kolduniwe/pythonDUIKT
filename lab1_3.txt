import math

def f(x: float) -> float:
    if x >= 0:
        # 0.5 - 4-й корінь з |x|
        return 0.5 - abs(x) ** 0.25
    else:
        denom = abs(x + 1)
        if denom == 0:
            raise ZeroDivisionError("Для x = -1 знаменник дорівнює нулю — значення невизначене.")
        # (sin(x^2))^2 / |x+1|
        return (math.sin(x * x) ** 2) / denom

# Ввід
try:
    x = float(input("Введіть x: "))
    y = f(x)
    # Вивід: чотири знаки після коми (можете змінити формат)
    print(f"f({x}) = {y:.4f}")
except ZeroDivisionError as e:
    print("Помилка:", e)
except ValueError:
    print("Невірний ввід. Введіть число.")
