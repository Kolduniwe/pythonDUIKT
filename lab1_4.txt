# Ввід даних
a = int(input("Введіть перше число: "))
b = int(input("Введіть друге число: "))
c = int(input("Введіть третє число: "))
N = int(input("Введіть N: "))

# Перевірка належності до інтервалу [1, N]
numbers = [a, b, c]
result = [x for x in numbers if 1 <= x <= N]

# Вивід результату
if result:
    print("Числа, що належать інтервалу [1, N]:", result)
else:
    print("Серед введених чисел немає тих, що належать інтервалу [1, N].")
