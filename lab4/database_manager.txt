import sqlite3

class DatabaseManager:
    def __init__(self, db_name="students.db"):
        self.conn = sqlite3.connect(db_name)
        self.cursor = self.conn.cursor()

    def create_table(self):
        self.cursor.execute('''CREATE TABLE IF NOT EXISTS students (
                                id INTEGER PRIMARY KEY AUTOINCREMENT,
                                pib TEXT,
                                group_number TEXT,
                                birth_date TEXT,
                                address TEXT,
                                avg_grade REAL,
                                desired_avg REAL)''')
        self.conn.commit()

    def insert_student(self, student, avg_grade, desired_avg):
        self.cursor.execute('''INSERT INTO students (pib, group_number, birth_date, address, avg_grade, desired_avg)
                               VALUES (?, ?, ?, ?, ?, ?)''',
                            (student.get_pib(), student.get_group(), student.get_birth_date(),
                             student.get_address(), avg_grade, desired_avg))
        self.conn.commit()

    def update_student(self, student_id, new_avg):
        self.cursor.execute('''UPDATE students SET avg_grade=? WHERE id=?''', (new_avg, student_id))
        self.conn.commit()

    def delete_student(self, student_id):
        self.cursor.execute('DELETE FROM students WHERE id=?', (student_id,))
        self.conn.commit()

    def show_all(self):
        self.cursor.execute('SELECT * FROM students')
        for row in self.cursor.fetchall():
            print(row)

    def __del__(self):
        self.conn.close()
